# Optional: semantic vector search.
#
# ChromaDB ships an ONNX build of all-MiniLM-L6-v2, so embeddings work with no
# extra download and no torch. It depends on onnxruntime, which does not always
# have wheels for the newest Python — hence the separate file, so `make setup`
# can continue without it.
#
# Without this, vector_memory uses its TF-IDF backend: no downloads, fully
# offline, but matching on wording rather than meaning.
#
# sentence-transformers is a further option (vector_memory prefers it when
# present) but pulls in ~1 GB of torch for the same model, so it is not listed.
chromadb>=0.5.0
